    <script>
      document.addEventListener("click", function (event) {
        var button = event.target.closest(".nav-toggle");
        if (!button) return;
        var open = document.body.classList.toggle("nav-open");
        button.setAttribute("aria-expanded", open ? "true" : "false");
      });
